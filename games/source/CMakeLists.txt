cmake_minimum_required(VERSION 3.22)
project(GeminiGamesNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Concurrent Gui Qml Quick Network)
add_library(GeminiGamesReview5 MODULE backend/Plugin.cpp backend/GeminiGamesBackend.cpp backend/GameArtwork.cpp backend/GameLibrary.cpp backend/RuntimeAdapters.cpp)
target_link_libraries(GeminiGamesReview5 PRIVATE Qt6::Core Qt6::Concurrent Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Network ${CMAKE_DL_LIBS})
target_compile_options(GeminiGamesReview5 PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(GeminiGamesReview5 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/native/review5")
add_executable(games-plugin-smoke backend/plugin_smoke.cpp)
target_link_libraries(games-plugin-smoke PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)

configure_file(qml/native/review5/qmldir native/review5/qmldir COPYONLY)
configure_file(scripts/games_manager.py management/games_manager.py COPYONLY)

find_package(Qt6 REQUIRED COMPONENTS Test)
add_library(GeminiGamesTestCore STATIC backend/GeminiGamesBackend.cpp backend/GameArtwork.cpp backend/GameLibrary.cpp backend/RuntimeAdapters.cpp)
target_include_directories(GeminiGamesTestCore PUBLIC backend)
target_link_libraries(GeminiGamesTestCore PUBLIC Qt6::Core Qt6::Concurrent Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Network)
add_executable(games-tests tests/games_test.cpp)
target_link_libraries(games-tests PRIVATE GeminiGamesTestCore Qt6::Test)
