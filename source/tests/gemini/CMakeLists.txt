cmake_minimum_required(VERSION 3.22)
project(GeminiRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network DBus Concurrent Quick Quick3D QuickTest)
set(SRC "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
add_executable(GeminiNetworkProbe network_probe.cpp "${SRC}/NetworkManager.cpp" "${SRC}/SecureIpcClient.cpp")
target_include_directories(GeminiNetworkProbe PRIVATE "${SRC}")
target_link_libraries(GeminiNetworkProbe PRIVATE Qt6::Core Qt6::Network Qt6::DBus Qt6::Concurrent)
add_executable(GeminiSetupProbe setup_probe.cpp "${SRC}/SetupWizardManager.cpp")
target_include_directories(GeminiSetupProbe PRIVATE "${SRC}")
target_link_libraries(GeminiSetupProbe PRIVATE Qt6::Core Qt6::Network)

add_executable(GeminiQmlRegression ControllerReplay.h ControllerReplay.cpp "${SRC}/ControllerManager.cpp" qml_runner.cpp "${SRC}/ClockSettingsManager.cpp" "${SRC}/ClockSettingsManager.h" "${SRC}/GeminiGeometry.cpp" "${SRC}/GeminiGeometry.h")
target_include_directories(GeminiQmlRegression PRIVATE "${SRC}")
target_link_libraries(GeminiQmlRegression PRIVATE Qt6::QuickTest Qt6::Quick3D Qt6::Quick Qt6::DBus Qt6::Network)

find_package(Qt6 REQUIRED COMPONENTS Test)
